name: Build APK
on:
  push:
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: npm install
      - run: npx cap add android
      - name: Mikrofon ruxsatini qo'shish
        run: sed -i 's|<application|<uses-permission android:name="android.permission.RECORD_AUDIO"/><uses-permission android:name="android.permission.MODIFY_AUDIO_SETTINGS"/><application|' android/app/src/main/AndroidManifest.xml
      - name: Doimiy imzo va versiya
        run: |
          cp keystore.p12 android/app/keystore.p12
          sed -i "s/versionCode 1/versionCode ${{ github.run_number }}/" android/app/build.gradle
          cat >> android/app/build.gradle <<'EOF2'
          android {
              signingConfigs {
                  fixed {
                      storeFile file("keystore.p12")
                      storePassword "organamiz123"
                      keyAlias "organamiz"
                      keyPassword "organamiz123"
                      storeType "pkcs12"
                  }
              }
              buildTypes {
                  debug {
                      signingConfig signingConfigs.fixed
                  }
              }
          }
          EOF2
      - run: cd android && chmod +x gradlew && ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: organamiz-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
